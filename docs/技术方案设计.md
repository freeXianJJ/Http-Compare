## 一、技术选型

### 1.1 核心技术栈

| 技术分类 | 选型 | 理由 |

| 前端框架 | **React 18 + TypeScript** | 生态成熟、类型安全、组件化开发效率高 |

### 3.2 请求管理器设计

```typescript
// services/requestService.ts

import axios, { AxiosInstance, AxiosRequestConfig } from 'axios';
import { ApiConfig, HttpMethod, ParamType } from '../types/api';
import { ServiceConfig } from '../types/service';
import { RequestInfo, ResponseInfo } from '../types/diff';

export class RequestService {
  private axiosInstance: AxiosInstance;

  constructor() {
    this.axiosInstance = axios.create({
      timeout: 30000,
      validateStatus: () => true // 不自动抛出错误
    });
  }

  /**
   * 构建完整URL
   */
  private buildUrl(
    service: ServiceConfig,
    apiConfig: ApiConfig
  ): string {
    const { protocol, host, port } = service;
    let url = apiConfig.url;

    // 替换路径参数
    const pathParams = apiConfig.parameters.filter(
      p => p.type === ParamType.PATH && p.enabled
    );
    pathParams.forEach(param => {
      url = url.replace(`{${param.name}}`, encodeURIComponent(param.value));
    });

    // 添加Query参数
    const queryParams = apiConfig.parameters.filter(
      p => p.type === ParamType.QUERY && p.enabled
    );
    if (queryParams.length > 0) {
      const queryString = queryParams
        .map(p => `${encodeURIComponent(p.name)}=${encodeURIComponent(p.value)}`)
        .join('&');
      url += `?${queryString}`;
    }

    return `${protocol}://${host}:${port}${url}`;
  }

  /**
   * 构建请求头
   */
  private buildHeaders(
    service: ServiceConfig,
    apiConfig: ApiConfig
  ): Record<string, string> {
    const headers: Record<string, string> = {};

    // 添加Token
    if (service.token) {
      const tokenValue = service.tokenPrefix 
        ? `${service.tokenPrefix} ${service.token}`
        : service.token;
      headers[service.tokenHeader] = tokenValue;
    }

    // 添加自定义Headers
    apiConfig.headers
      .filter(h => h.enabled)
      .forEach(h => {
        headers[h.name] = h.value;
      });

    // 根据请求体类型设置Content-Type
    const bodyParams = apiConfig.parameters.filter(
      p => (p.type === ParamType.BODY || p.type === ParamType.FORM) && p.enabled
    );
    if (bodyParams.length > 0) {
      const isJson = bodyParams.some(p => p.type === ParamType.BODY);
      headers['Content-Type'] = isJson
        ? 'application/json'
        : 'application/x-www-form-urlencoded';
    }

    return headers;
  }

  /**
   * 构建请求体
   */
  private buildBody(apiConfig: ApiConfig): any {
    const bodyParams = apiConfig.parameters.filter(
      p => p.type === ParamType.BODY && p.enabled
    );
    const formParams = apiConfig.parameters.filter(
      p => p.type === ParamType.FORM && p.enabled
    );

    if (bodyParams.length > 0) {
      // JSON Body
      try {
        // 假设第一个body参数的value是完整的JSON字符串
        return JSON.parse(bodyParams[0].value);
      } catch (e) {
        // 如果解析失败，尝试构建对象
        const body: Record<string, any> = {};
        bodyParams.forEach(p => {
          try {
            body[p.name] = JSON.parse(p.value);
          } catch {
            body[p.name] = p.value;
          }
        });
        return body;
      }
    }

    if (formParams.length > 0) {
      // Form Data
      const formData = new URLSearchParams();
      formParams.forEach(p => {
        formData.append(p.name, p.value);
      });
      return formData.toString();
    }

    return undefined;
  }

  /**
   * 发送请求
   */
  async sendRequest(
    service: ServiceConfig,
    apiConfig: ApiConfig
  ): Promise<{ request: RequestInfo; response: ResponseInfo }> {
    const startTime = Date.now();
    const url = this.buildUrl(service, apiConfig);
    const headers = this.buildHeaders(service, apiConfig);
    const body = this.buildBody(apiConfig);

    const requestInfo: RequestInfo = {
      url,
      method: apiConfig.method,
      headers,
      body,
      timestamp: startTime,
      duration: 0
    };

    try {
      const config: AxiosRequestConfig = {
        method: apiConfig.method,
        url,
        headers,
        data: body
      };

      const response = await this.axiosInstance.request(config);
      const endTime = Date.now();

      requestInfo.duration = endTime - startTime;

      const responseInfo: ResponseInfo = {
        status: response.status,
        statusText: response.statusText,
        headers: response.headers as Record<string, string>,
        body: response.data,
        timestamp: endTime
      };

      return { request: requestInfo, response: responseInfo };
    } catch (error: any) {
      const endTime = Date.now();
      requestInfo.duration = endTime - startTime;

      const responseInfo: ResponseInfo = {
        status: 0,
        statusText: 'Error',
        headers: {},
        body: null,
        timestamp: endTime,
        error: error.message
      };

      return { request: requestInfo, response: responseInfo };
    }
  }

  /**
   * 并发发送双端请求
   */
  async sendDualRequests(
    oldService: ServiceConfig,
    newService: ServiceConfig,
    apiConfig: ApiConfig
  ): Promise<{
    oldService: { request: RequestInfo; response: ResponseInfo };
    newService: { request: RequestInfo; response: ResponseInfo };
  }> {
    const [oldResult, newResult] = await Promise.all([
      this.sendRequest(oldService, apiConfig),
      this.sendRequest(newService, apiConfig)
    ]);

    return {
      oldService: oldResult,
      newService: newResult
    };
  }
}
```

### 3.3 差异对比引擎设计

```typescript
// services/diffService.ts

import * as deepDiff from 'deep-diff';
import { DiffResult, DiffItem, ResponseInfo } from '../types/diff';
import { compareArraysUnordered } from '../utils/arrayCompare';

export class DiffService {
  /**
   * 对比两个响应
   */
  compare(
    oldResponse: ResponseInfo,
    newResponse: ResponseInfo
  ): DiffResult {
    const statusMatch = oldResponse.status === newResponse.status;
    const headerMatch = this.compareHeaders(oldResponse.headers, newResponse.headers);
    const bodyDiffs = this.compareBody(oldResponse.body, newResponse.body);

    return {
      identical: statusMatch && headerMatch && bodyDiffs.length === 0,
      differences: [
        ...this.getStatusDiffs(oldResponse, newResponse),
        ...bodyDiffs
      ],
      summary: {
        statusMatch,
        headerMatch,
        bodyMatch: bodyDiffs.length === 0,
        totalDiffs: bodyDiffs.length + (statusMatch ? 0 : 1)
      }
    };
  }

  /**
   * 对比状态码
   */
  private getStatusDiffs(
    oldResponse: ResponseInfo,
    newResponse: ResponseInfo
  ): DiffItem[] {
    if (oldResponse.status === newResponse.status) {
      return [];
    }

    return [{
      path: 'status',
      type: 'modified',
      oldValue: oldResponse.status,
      newValue: newResponse.status
    }];
  }

  /**
   * 对比响应头
   */
  private compareHeaders(
    oldHeaders: Record<string, string>,
    newHeaders: Record<string, string>
  ): boolean {
    // 简化处理，只对比关键头
    const keyHeaders = ['content-type', 'content-length'];
    for (const key of keyHeaders) {
      if (oldHeaders[key] !== newHeaders[key]) {
        return false;
      }
    }
    return true;
  }

  /**
   * 对比响应体
   */
  private compareBody(oldBody: any, newBody: any): DiffItem[] {
    const diffs: DiffItem[] = [];

    try {
      this.deepCompare(oldBody, newBody, '', diffs);
    } catch (error) {
      console.error('对比出错:', error);
    }

    return diffs;
  }

  /**
   * 深度对比对象
   */
  private deepCompare(
    oldVal: any,
    newVal: any,
    path: string,
    diffs: DiffItem[]
  ): void {
    // 类型检查
    const oldType = this.getType(oldVal);
    const newType = this.getType(newVal);

    // 类型不一致
    if (oldType !== newType) {
      diffs.push({
        path: path || 'root',
        type: 'type-mismatch',
        oldValue: oldVal,
        newValue: newVal,
        oldType,
        newType
      });
      return;
    }

    // null或undefined
    if (oldVal === null || oldVal === undefined) {
      if (oldVal !== newVal) {
        diffs.push({
          path: path || 'root',
          type: 'modified',
          oldValue: oldVal,
          newValue: newVal
        });
      }
      return;
    }

    // 数组对比（无序）
    if (Array.isArray(oldVal)) {
      this.compareArrays(oldVal, newVal, path, diffs);
      return;
    }

    // 对象对比
    if (oldType === 'object') {
      this.compareObjects(oldVal, newVal, path, diffs);
      return;
    }

    // 基本类型对比（严格相等）
    if (oldVal !== newVal) {
      diffs.push({
        path: path || 'root',
        type: 'modified',
        oldValue: oldVal,
        newValue: newVal
      });
    }
  }

  /**
   * 对比数组（无序）
   */
  private compareArrays(
    oldArr: any[],
    newArr: any[],
    path: string,
    diffs: DiffItem[]
  ): void {
    // 如果数组长度不同，先记录
    if (oldArr.length !== newArr.length) {
      // 继续对比，但已经有差异了
    }

    // 简化处理：如果数组元素是基本类型，使用Set对比
    if (oldArr.length > 0 && this.isPrimitive(oldArr[0])) {
      const oldSet = new Set(oldArr.map(v => JSON.stringify(v)));
      const newSet = new Set(newArr.map(v => JSON.stringify(v)));

      const added = [...newSet].filter(v => !oldSet.has(v));
      const deleted = [...oldSet].filter(v => !newSet.has(v));

      if (added.length > 0 || deleted.length > 0) {
        diffs.push({
          path: path,
          type: 'modified',
          oldValue: oldArr,
          newValue: newArr
        });
      }
    } else {
      // 对象数组：尝试逐个匹配
      const matched = new Set<number>();
      
      for (let i = 0; i < oldArr.length; i++) {
        let found = false;
        for (let j = 0; j < newArr.length; j++) {
          if (matched.has(j)) continue;
          
          const tempDiffs: DiffItem[] = [];
          this.deepCompare(oldArr[i], newArr[j], '', tempDiffs);
          
          if (tempDiffs.length === 0) {
            matched.add(j);
            found = true;
            break;
          }
        }
        
        if (!found) {
          diffs.push({
            path: `${path}[${i}]`,
            type: 'modified',
            oldValue: oldArr[i],
            newValue: undefined
          });
        }
      }

      // 检查新数组中未匹配的元素
      for (let j = 0; j < newArr.length; j++) {
        if (!matched.has(j)) {
          diffs.push({
            path: `${path}[${j}]`,
            type: 'added',
            oldValue: undefined,
            newValue: newArr[j]
          });
        }
      }
    }
  }

  /**
   * 对比对象
   */
  private compareObjects(
    oldObj: Record<string, any>,
    newObj: Record<string, any>,
    path: string,
    diffs: DiffItem[]
  ): void {
    const allKeys = new Set([
      ...Object.keys(oldObj),
      ...Object.keys(newObj)
    ]);

    for (const key of allKeys) {
      const newPath = path ? `${path}.${key}` : key;
      const oldHas = key in oldObj;
      const newHas = key in newObj;

      if (!oldHas) {
        diffs.push({
          path: newPath,
          type: 'added',
          oldValue: undefined,
          newValue: newObj[key]
        });
      } else if (!newHas) {
        diffs.push({
          path: newPath,
          type: 'deleted',
          oldValue: oldObj[key],
          newValue: undefined
        });
      } else {
        this.deepCompare(oldObj[key], newObj[key], newPath, diffs);
      }
    }
  }

  /**
   * 获取值的类型
   */
  private getType(val: any): string {
    if (val === null) return 'null';
    if (val === undefined) return 'undefined';
    if (Array.isArray(val)) return 'array';
    return typeof val;
  }

  /**
   * 判断是否为基本类型
   */
  private isPrimitive(val: any): boolean {
    return val === null || 
           val === undefined ||
           typeof val === 'string' ||
           typeof val === 'number' ||
           typeof val === 'boolean';
  }
}
```

### 3.4 状态管理设计

```typescript
// stores/apiStore.ts

import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import { ApiConfig, HttpMethod, Parameter, ParamType } from '../types/api';
import { nanoid } from 'nanoid';

interface ApiState {
  currentApi: ApiConfig | null;
  savedApis: ApiConfig[];
  
  // Actions
  setCurrentApi: (api: ApiConfig) => void;
  updateCurrentApi: (updates: Partial<ApiConfig>) => void;
  addParameter: (param: Omit<Parameter, 'id'>) => void;
  updateParameter: (id: string, updates: Partial<Parameter>) => void;
  removeParameter: (id: string) => void;
  saveCurrentApi: () => void;
  loadApi: (id: string) => void;
  deleteApi: (id: string) => void;
  createNewApi: () => void;
}

export const useApiStore = create<ApiState>()(
  persist(
    (set, get) => ({
      currentApi: null,
      savedApis: [],

      setCurrentApi: (api) => set({ currentApi: api }),

      updateCurrentApi: (updates) => set((state) => ({
        currentApi: state.currentApi
          ? { ...state.currentApi, ...updates, updatedAt: Date.now() }
          : null
      })),

      addParameter: (param) => set((state) => {
        if (!state.currentApi) return state;
        
        const newParam: Parameter = {
          ...param,
          id: nanoid()
        };

        return {
          currentApi: {
            ...state.currentApi,
            parameters: [...state.currentApi.parameters, newParam],
            updatedAt: Date.now()
          }
        };
      }),

      updateParameter: (id, updates) => set((state) => {
        if (!state.currentApi) return state;

        return {
          currentApi: {
            ...state.currentApi,
            parameters: state.currentApi.parameters.map(p =>
              p.id === id ? { ...p, ...updates } : p
            ),
            updatedAt: Date.now()
          }
        };
      }),

      removeParameter: (id) => set((state) => {
        if (!state.currentApi) return state;

        return {
          currentApi: {
            ...state.currentApi,
            parameters: state.currentApi.parameters.filter(p => p.id !== id),
            updatedAt: Date.now()
          }
        };
      }),

      saveCurrentApi: () => set((state) => {
        if (!state.currentApi) return state;

        const existingIndex = state.savedApis.findIndex(
          api => api.id === state.currentApi!.id
        );

        if (existingIndex >= 0) {
          // 更新已存在的接口
          const newSavedApis = [...state.savedApis];
          newSavedApis[existingIndex] = state.currentApi;
          return { savedApis: newSavedApis };
        } else {
          // 添加新接口
          return {
            savedApis: [...state.savedApis, state.currentApi]
          };
        }
      }),

      loadApi: (id) => set((state) => {
        const api = state.savedApis.find(a => a.id === id);
        return api ? { currentApi: { ...api } } : state;
      }),

      deleteApi: (id) => set((state) => ({
        savedApis: state.savedApis.filter(a => a.id !== id),
        currentApi: state.currentApi?.id === id ? null : state.currentApi
      })),

      createNewApi: () => set({
        currentApi: {
          id: nanoid(),
          name: '新建接口',
          url: '',
          method: HttpMethod.GET,
          parameters: [],
          headers: [],
          createdAt: Date.now(),
          updatedAt: Date.now()
        }
      })
    }),
    {
      name: 'api-storage',
      partialize: (state) => ({
        savedApis: state.savedApis
      })
    }
  )
);
```

```typescript
// App.tsx

      };

      addToHistory(testResult);
    } catch (error) {
      message.error('请求失败: ' + error.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <Layout style={{ height: '100vh' }}>
      <Header style={{ 
        background: '#fff',
```

### 4.2 差异展示组件

```typescript
// components/ResponseDiff/JsonDiff.tsx

import React from 'react';
import { DiffItem } from '../../types/diff';
import './JsonDiff.css';

interface JsonDiffProps {
  oldData: any;
  newData: any;
  differences: DiffItem[];
}

export const JsonDiff: React.FC<JsonDiffProps> = ({
  oldData,
  newData,
  differences
}) => {
  const diffMap = new Map<string, DiffItem>();
  differences.forEach(d => diffMap.set(d.path, d));

  const renderValue = (
    value: any,
    path: string,
    side: 'old' | 'new'
  ): React.ReactNode => {
    const diff = diffMap.get(path);
    
    if (value === null || value === undefined) {
      return (
        <span className={getDiffClass(diff, side)}>
          {String(value)}
        </span>
      );
    }

    if (Array.isArray(value)) {
      return (
        <div className="json-array">
          [
          {value.map((item, idx) => (
            <div key={idx} className="json-array-item">
              {renderValue(item, `${path}[${idx}]`, side)},
            </div>
          ))}
          ]
        </div>
      );
    }

    if (typeof value === 'object') {
      return (
        <div className="json-object">
          {'{'}
          {Object.entries(value).map(([key, val]) => (
            <div key={key} className="json-object-item">
              <span className="json-key">"{key}"</span>: 
              {renderValue(val, path ? `${path}.${key}` : key, side)}
            </div>
          ))}
          {'}'}
        </div>
      );
    }

    return (
      <span className={getDiffClass(diff, side)}>
        {typeof value === 'string' ? `"${value}"` : String(value)}
      </span>
    );
  };

  const getDiffClass = (diff: DiffItem | undefined, side: 'old' | 'new'): string => {
    if (!diff) return 'json-value-identical';
    
    if (diff.type === 'added') {
      return side === 'new' ? 'json-value-added' : 'json-value-missing';
    }
    
    if (diff.type === 'deleted') {
      return side === 'old' ? 'json-value-deleted' : 'json-value-missing';
    }
    
    return 'json-value-modified';
  };

  return (
    <div className="json-diff-container">
      <div className="json-diff-panel">
        <div className="json-diff-header">旧服务响应</div>
        <div className="json-diff-content">
          {renderValue(oldData, '', 'old')}
        </div>
      </div>
      
      <div className="json-diff-panel">
        <div className="json-diff-header">新服务响应</div>
        <div className="json-diff-content">
          {renderValue(newData, '', 'new')}
        </div>
      </div>
    </div>
  );
};
```

## 八、总结

本技术方案采用现代化的前端技术栈，以React + TypeScript + Ant Design为核心，实现了一个纯前端的HTTP接口对比测试工具。方案的主要特点包括：

1. **架构清晰**：分层设计，职责明确，易于维护和扩展
2. **类型安全**：全面使用TypeScript，减少运行时错误
3. **状态管理**：使用Zustand轻量级状态管理，支持持久化
4. **对比准确**：严格类型对比、数组无序对比、深度对象对比
5. **部署简单**：纯静态文件，支持Nginx直接托管
6. **扩展性好**：为批量测试、插件系统预留接口

该方案能够满足当前需求，并为未来功能扩展打下良好基础。
